function replay_mixed(root,work)
previous=pwd;
cd(fullfile(root,'matlab/active'));
setenv('ACE_INPUT_MODE','shared');
setenv('ACE_VALIDATE_INPUT',fullfile(work,'matlab_replay_init.mat'));
ace;
setenv('ACE_VALIDATE_INPUT','');
global one_ace_only T Q QL QI
one_ace_only=true;
fid=fopen(fullfile(work,'states.bin'),'r','ieee-le');
states=reshape(fread(fid,8*801*3*60,'double'),[8,801,3,60]); fclose(fid);
fid=fopen(fullfile(work,'matlab.bin'),'w','ieee-le');
cache=fopen(fullfile(work,'matlab_thermo.bin'),'w','ieee-le');
for sample=1:60
 state=zeros(Nzm,5,Ni);
 state(:,1:3,:)=permute(states(:,:,:,sample),[2,3,1]);
 raw=reshape(expFun(state(:),1),[Nzm,5,Ni]);
 thermo=permute(cat(3,T,Q,QL,QI),[1,3,2]);
 fwrite(cache,permute(thermo,[3,1,2]),'double');
 [dth,dqt]=PRECIP_SINKS(T,Q,QL,QI);
 sink=permute(cat(3,dth,dqt),[1,3,2]);
 qc=QL+QI; dql=1e-8*QL./qc; dqi=1e-8*QI./qc;
 dql(qc<=0)=0; dqi(qc<=0)=0; % diagnostic forward values only
 forward=permute(cat(3,eval_theta_e(T,Q,QL,QI,p_zc),...
   eval_theta_e(T,Q,QL+dql,QI+dqi,p_zc)),[1,3,2]);
 dynamics=SSPRK93(state(:),dt,[]);
 after=ApplyPrecip(dynamics,dt);
 shaped=reshape(after,[Nzm,5,Ni]);
 shaped(:,2,:)=shaped(:,2,:).*(shaped(:,2,:)>0);
 final=reshape(ImDamp(shaped(:)),[Nzm,5,Ni]);
 dynamics=reshape(dynamics,[Nzm,5,Ni]); after=shaped;
 % Match the Fortran stream layout, including the shared-cache sink slots.
 blocks={raw(:,1:3,:),thermo,sink,forward,sink,forward,...
         dynamics(:,1:3,:),after(:,1:3,:),final(:,1:3,:)};
 for j=1:numel(blocks)
  fwrite(fid,permute(blocks{j},[3,1,2]),'double');
 end
end
fclose(fid); fclose(cache);
cd(previous);
end
